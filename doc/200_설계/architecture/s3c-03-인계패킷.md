### 1.3 공개 시그니처 (contract가 전제로 쓰는 것 — 바꾸면 contract에 통보)

```ts
// server/src/auth
export const requireOwner: MiddlewareHandler<AppEnv>   // requireToken 뒤. 아니면 AppError('OWNER_ONLY')
// AuthService 에 추가
isOwner: (principal: Principal) => boolean

// server/src/settings
export type SettingsService = {
  /** D1 행 → 재검증 → 응답형. 행 없음·재검증 실패면 시드(isDefault: true, version 0, updatedAt null) */
  get: () => Promise<CharacterSettingsResponse>
  /** 입력은 routes 가 zod 로 이미 검증한 값. 정규화(trim·빈 항목 제거) → UPSERT → 응답형 */
  put: (settings: CharacterSettings, by: Principal) => Promise<CharacterSettingsResponse>
  /** speak·regenerate 용. get 과 같은 출처, 프롬프트 입력형으로 */
  loadForPrompt: () => Promise<{ profiles: Readonly<Record<CharacterId, CharacterProfile>>; common: CommonPrompt }>
}
// zod 스키마 위치: server/src/settings/schema.ts 의 characterSettingsSchema — routes/schemas.ts 가 재사용(정의 1곳)
```

`CharacterSettings`·`CharacterSettingsResponse` 타입은 `@shared/types`(contract 소유)에서 import한다. 4단계(shared)가 5단계보다 먼저다.
